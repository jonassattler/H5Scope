# The build for xcb-util-cursor, copied into the extracted tarball by the
# portfile beside this file. See that file for why upstream's own autotools
# build is not used.
#
# What this has to reproduce faithfully is what `configure` puts on the compile
# line, and that is a shorter list than it looks: configure.ac has no
# AC_CONFIG_HEADERS, so there is no config.h anywhere in the sources and every
# AC_DEFINE arrives as a -D. Two of them are load-bearing and both fail loudly
# rather than quietly if they are wrong -- a missing HAVE_ENDIAN_H leaves
# le32toh undeclared, and a missing XCURSORPATH leaves load_cursor.c with an
# undefined identifier. There is no third.

cmake_minimum_required(VERSION 3.20)
project(xcb-util-cursor C)

include(CheckIncludeFile)

# The headers this compiles against belong to three libraries that stay the
# host's: xcb, xcb-render and xcb-image (xcb-renderutil's header comes with the
# last). Only the headers are needed here -- a static archive links nothing --
# so this is a compile-time requirement on the build machine and not a runtime
# one on the target, where all three have been in RHEL 8's own repositories all
# along. Four find_path calls rather than pkg-config, because pkg-config inside
# a vcpkg port build is pointed at vcpkg's prefix and these are not there.
set(xcb_include_dirs "")

function(require_xcb_header header debian rhel)
    string(MAKE_C_IDENTIFIER "${header}" slug)
    find_path(XCB_CURSOR_DIR_${slug} NAMES "${header}")
    if(NOT XCB_CURSOR_DIR_${slug})
        message(FATAL_ERROR
            "<${header}> was not found.\n"
            "xcb-util-cursor compiles against the system's X headers. Install "
            "'${rhel}' on RHEL and its relatives, or '${debian}' on Debian and "
            "Ubuntu.")
    endif()
    set(xcb_include_dirs ${xcb_include_dirs} "${XCB_CURSOR_DIR_${slug}}"
        PARENT_SCOPE)
endfunction()

require_xcb_header("xcb/xcb.h"            "libxcb1-dev"             "libxcb-devel")
require_xcb_header("xcb/render.h"         "libxcb-render0-dev"      "libxcb-devel")
require_xcb_header("xcb/xcb_renderutil.h" "libxcb-render-util0-dev" "xcb-util-renderutil-devel")
require_xcb_header("xcb/xcb_image.h"      "libxcb-image0-dev"       "xcb-util-image-devel")
list(REMOVE_DUPLICATES xcb_include_dirs)

# configure.ac probes four spellings of the endianness header and stops at the
# first that exists. This port only ever builds on Linux -- the portfile says
# so and refuses otherwise -- so that first one is always <endian.h>, and the
# probe is kept only to fail here, with a sentence, rather than inside
# parse_cursor_file.c with `le32toh` undeclared.
check_include_file("endian.h" HAVE_ENDIAN_H)
if(NOT HAVE_ENDIAN_H)
    message(FATAL_ERROR
        "<endian.h> was not found. parse_cursor_file.c needs le32toh and has "
        "no fallback that applies on Linux.")
endif()

# configure's own default for --with-cursorpath, verbatim. It is the search
# path for cursor themes, so a wrong value here is a binary whose cursors
# silently fall back to the ugly X11 core font.
set(XCURSORPATH "~/.icons:/usr/share/icons:/usr/share/pixmaps:/usr/X11R6/lib/X11/icons")

# shape_to_id.c is generated by gperf, and comes pre-generated in the release
# tarball -- so gperf is not a build requirement here.
#
# One consequence of linking this statically rather than as a shared object.
# Three of its symbols are neither `static` nor named xcb_cursor_* --
# `in_word_set` (gperf's), `cursor_shape_to_id` and `parse_cursor_file` -- and
# the shared build hides them behind -export-symbols-regex '^xcb_cursor_',
# which an archive has no equivalent of. Nothing else on H5Scope's link line
# defines any of the three; it was checked against every archive in the vcpkg
# tree. Were that to change, the link would fail with "multiple definition" and
# name the symbol, which is the failure mode to want.
add_library(xcb-cursor STATIC
    cursor/cursor.c
    cursor/shape_to_id.c
    cursor/load_cursor.c
    cursor/parse_cursor_file.c
)

target_include_directories(xcb-cursor PRIVATE cursor ${xcb_include_dirs})

target_compile_definitions(xcb-cursor PRIVATE
    # AC_USE_SYSTEM_EXTENSIONS, which is what puts le32toh in <endian.h>.
    _GNU_SOURCE
    HAVE_ENDIAN_H
    "XCURSORPATH=\"${XCURSORPATH}\""
)

# The archive is linked into a PIE executable, so every object in it has to be
# position-independent. vcpkg's Linux triplets already compile with -fPIC; this
# says it in the build itself, where a reader can see it.
set_target_properties(xcb-cursor PROPERTIES POSITION_INDEPENDENT_CODE ON)

# Plain `lib` and `include`, not GNUInstallDirs: vcpkg's layout is the same on
# every distribution, and a libdir that came out as `lib64` on RHEL would fail
# its post-build checks.
install(TARGETS xcb-cursor ARCHIVE DESTINATION lib)
install(FILES cursor/xcb_cursor.h DESTINATION include/xcb)
