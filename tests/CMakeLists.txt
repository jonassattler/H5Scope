add_library(test_support STATIC support/TestFile.cpp)
target_include_directories(test_support PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(test_support PUBLIC HDF5::HDF5)

# The entry point for the suites that reach HDF5 through the application. See
# the header of CatchMain.cpp: the HDF5 thread runs an event loop, and an event
# loop needs a QCoreApplication to run in.
add_library(test_main STATIC support/CatchMain.cpp)
target_link_libraries(test_main PUBLIC gui Catch2::Catch2 Qt6::Core)

include(Catch)

# --- backend: pure C++, no Qt --------------------------------------------
add_executable(test_h5core test_h5core.cpp)
target_link_libraries(test_h5core PRIVATE h5core test_support Catch2::Catch2WithMain)
catch_discover_tests(test_h5core)

# --- the postprocessing pipeline: Qt Core for the shared subscript grammar,
# --- but no models, no controller and no engine ---------------------------
# tests/data/NumpyGolden.hpp is generated by tools/make-numpy-golden.py and
# committed, so this needs no interpreter and no numpy to run -- see the header
# of that script for why it is not generated at build time.
add_executable(test_postprocess test_postprocess.cpp)
target_link_libraries(test_postprocess PRIVATE postproc Catch2::Catch2WithMain
                                              Qt6::Core)
catch_discover_tests(test_postprocess)

# --- the HDF5 thread: the rule that HDF5 is touched from one thread only ---
# Its own binary, and deliberately so. This suite asserts what is true before
# any file is open and what is true of a process where nothing has claimed the
# library yet, and both of those stop being observable the moment another suite
# in the same binary has started the thread.
add_executable(test_h5thread test_h5thread.cpp)
target_link_libraries(test_h5thread PRIVATE gui test_support test_main)
catch_discover_tests(test_h5thread)

# --- models and controller: Qt types, still no QML engine -----------------
add_executable(test_models test_models.cpp)
target_link_libraries(test_models PRIVATE gui test_support test_main Qt6::Test)
catch_discover_tests(test_models)

# --- the example file, read back through the application's own stack -------
# Shares tools/ExampleFile.cpp with make-example-file, so the suite asserts
# against exactly the file that program writes rather than a second copy of it.
add_executable(test_example test_example.cpp)
target_link_libraries(test_example PRIVATE
  gui example_file test_support test_main Qt6::Test
)
catch_discover_tests(test_example)

# --- QML: the real bindings, via Qt Quick Test ----------------------------
qt_add_executable(test_qml tst_qml.cpp)
target_link_libraries(test_qml PRIVATE
  Qt6::QuickTest Qt6::Qml Qt6::Quick
  gui appqml test_support
  # PlotSurface.qml imports QtGraphs, and a static Qt resolves QML modules at
  # link time; qt_import_qml_plugins below scans this target's own QML files,
  # which are the test suites rather than the views they instantiate.
  Qt6::Graphs
)
target_compile_definitions(test_qml PRIVATE
  QUICK_TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/qml"
)
# A static Qt links QML modules rather than discovering them at run time.
qt_import_qml_plugins(test_qml)
# ...and only the default platform plugin, so headless runs need offscreen
# linked in explicitly or the binary aborts before any test executes.
qt_import_plugins(test_qml INCLUDE Qt6::QOffscreenIntegrationPlugin)

add_test(NAME qml COMMAND test_qml)
# The suites run in parallel (see CMakePresets.json), and these two are the
# only ones that could mind. Both drive a real scene graph and assert on what
# was rendered, with timeouts measured in wall-clock seconds; two of them on a
# two-core CI runner would be competing for the machine that decides whether
# they pass. The lock keeps them off each other while everything else -- which
# is pure CPU and has no such timing -- parallelises freely.
set_tests_properties(qml PROPERTIES
  ENVIRONMENT "QT_QPA_PLATFORM=offscreen"
  RESOURCE_LOCK rendering
)

# --- the QML root, loaded exactly as main.cpp does it ---------------------
qt_add_executable(test_qmlload tst_qmlload.cpp)
target_link_libraries(test_qmlload PRIVATE
  Qt6::Test Qt6::Qml Qt6::Quick Qt6::QuickControls2
  gui appqml appqmlplugin
)
# Main.qml reaches QtGraphs through DataView; a static Qt needs the module
# named or the root fails to load with 'module "QtGraphs" is not installed'.
target_link_libraries(test_qmlload PRIVATE Qt6::Graphs)
qt_import_qml_plugins(test_qmlload)
qt_import_plugins(test_qmlload INCLUDE Qt6::QOffscreenIntegrationPlugin)

add_test(NAME qmlload COMMAND test_qmlload)
set_tests_properties(qmlload PROPERTIES
  ENVIRONMENT "QT_QPA_PLATFORM=offscreen"
  RESOURCE_LOCK rendering
)

# --- design-system adherence ----------------------------------------------
# The portable half of the TBD design system's adherence rules: no raw hex
# colours and no font families outside src/qml/Theme.qml. Needs no build
# artefacts, so it runs first and fails fast.
add_test(NAME design_tokens
  COMMAND bash ${CMAKE_SOURCE_DIR}/tools/check-design-tokens.sh
               ${CMAKE_SOURCE_DIR}/src/qml
)

# Nothing elides without somewhere to read it in full. The rule is that data is
# never inaccessible for want of room, and an elided Text is the shape that
# breaks it; see the header of the script.
add_test(NAME elided_text
  COMMAND bash ${CMAKE_SOURCE_DIR}/tools/check-elided-text.sh
               ${CMAKE_SOURCE_DIR}/src/qml
)
